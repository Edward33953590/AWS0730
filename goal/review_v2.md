**实现要点**:
```python
# 数据埋点模型
class RecommendationLog(db.Model):
    """推荐日志 - 用于统计 AI vs 热门券效果"""
    id = db.Column(db.String(36), primary_key=True)
    user_id = db.Column(db.String(36), db.ForeignKey('users.id'))
    recommend_type = db.Column(db.String(20))  # 'ai' or 'popular'
    coupon_id = db.Column(db.String(36), db.ForeignKey('campaigns.id'))
    is_clicked = db.Column(db.Boolean, default=False)
    is_claimed = db.Column(db.Boolean, default=False)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)

# API 接口
GET /api/stats/ai-impact  # AI 效果统计数据
POST /api/ai/recommend/click  # 追踪推荐点击
```
